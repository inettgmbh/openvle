import { translate } from '@docusaurus/Translate'

export const FieldReference = ({ fields }) => (
  <table>
    <thead>
      <tr>
        <th>
          {translate({
            id: 'component.fieldReference.header.name',
            message: 'Feldname',
          })}
        </th>
        <th>
          {translate({
            id: 'component.fieldReference.header.description',
            message: 'Beschreibung',
          })}
        </th>
      </tr>
    </thead>
    <tbody>
      {fields
        .slice()
        .sort((a, b) => a.name.localeCompare(b.name))
        .map((field, index) => (
          <tr key={index}>
            <td>
              <code>{field.name}</code>
            </td>
            <td>
              {field.deprecated && (
                <span
                  title={translate({
                    id: 'component.fieldReference.deprecated.tooltip',
                    message:
                      'Dieses Feld ist veraltet und sollte nicht mehr verwendet werden.',
                  })}
                  style={{
                    backgroundColor: '#f4cccc',
                    color: '#a61c00',
                    borderRadius: '6px',
                    padding: '2px 6px',
                    fontSize: '0.85em',
                    marginRight: '6px',
                    fontWeight: 600,
                    cursor: 'help',
                  }}
                >
                  Deprecated
                </span>
              )}
              {field.description}
            </td>
          </tr>
        ))}
    </tbody>
  </table>
)
