import Admonition from '@theme/Admonition'

  return (
    <>
      <p>
        {translate({
          id: 'component.permissions.intro',
          message:
            'Die für Aktionen benötigten Berechtigungen können über Rollen oder individuell vergeben werden. Fehlen Ihnen bestimmte Rechte, sind die entsprechenden Funktionen in der Benutzeroberfläche ausgeblendet oder deaktiviert.',
        })}
      </p>
      <table>
        <thead>
          <tr>
            <th>
              {translate({
                id: 'component.permissions.header.action',
                message: 'Aktion',

            </th>
            <th>
              {translate({
                id: 'component.permissions.header.permission',
                message: 'Benötigte Berechtigung',
              })}
            </th>
            <th>
              {translate({
                id: 'component.permissions.header.path',
                message: 'Pfad',
              })}
            </th>
            <th>
              {translate({
                id: 'component.permissions.header.info',
                message: 'Weitere Informationen',
              })}
            </th>
          </tr>
        </thead>
        <tbody>
          {data.map(
            ({
              action,
              permission,

                <td>{action}</td>
                <td>
                  <code>{permission}</code>

                  ) : (

                        borderRadius: '6px',

                        fontWeight: 600,

                      }}
                    >
                      Deprecated
                    </span>
                  )}

                    <span
                      title={translate(

                          message: 'Eingeführt in Version {version}',

                        color: '#084298',
                        borderRadius: '6px',
                        padding: '2px 6px',
                        fontSize: '0.85em',

                    >
                      Version: {version}
                    </span>
                  )}
                  {beta && (
                    <span
                      title={translate({
                        id: 'component.permissions.beta.tooltip',
                        message:
                          'Diese Berechtigung sollte aktuell noch nicht verwendet werden.',

                        color: '#1E3A8A',
                        padding: '2px 6px',
                        fontSize: '0.85em',
                        marginRight: '6px',
                        fontWeight: 600,
                        cursor: 'help',
                      }}
                    >
                      Beta

          })}

            id: 'component.permissions.admonition.global',
            message:
              'Wird als Pfad / verwendet, gilt die Berechtigung global für alle Objekte dieses Typs.',
          })}
          <br />
          {translate({
            id: 'component.permissions.admonition.create',
            message:
              'Für *_create-Berechtigungen ist nur der globale Pfad / zulässig.',
          })}

      )}
    </>
  )
